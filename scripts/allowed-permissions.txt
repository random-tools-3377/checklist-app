# Every permission the release APK may ask for. Anything else fails CI.
# Network permissions are never allowed, whatever this file says.

# Added by AndroidX so the app's own broadcast receivers are private to it. Not a network permission.
${applicationId}.DYNAMIC_RECEIVER_NOT_EXPORTED_PERMISSION

# Notifications at set times (0.3.0), all scheduled on the phone.
android.permission.POST_NOTIFICATIONS
android.permission.USE_EXACT_ALARM
android.permission.SCHEDULE_EXACT_ALARM
android.permission.RECEIVE_BOOT_COMPLETED
# Added by WorkManager, which the home-screen widget library runs on. Not a network permission.
android.permission.WAKE_LOCK

# The privacy lock (0.4.0) asks for the phone's own fingerprint, face or PIN.
android.permission.USE_BIOMETRIC
android.permission.USE_FINGERPRINT

# The phone's own calendar (0.32.0), read on the phone only, once the user turns it on in Settings.
android.permission.READ_CALENDAR
